# Save Drawing as PNG Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a dock Save button that downloads a PNG of all board strokes, cropped to content with paper background and ~48px padding.

**Architecture:** Pure geometry helpers in a small Node-importable module compute content bounds and export canvas size. The browser gathers live strokes, draws them onto an offscreen canvas (same pen/eraser rules as the live board), and triggers a PNG download. No server or WebSocket changes.

**Tech Stack:** Plain HTML/Canvas JS, Node `node:test` for bbox helpers, existing dock UI patterns.

## Global Constraints

- Export content: completed + in-progress local + in-progress remote strokes
- Crop to stroke bounds (points ± half stroke `width`), then **48px** padding
- Background solid `#eef2f6` (app `--paper`); no grid, dock, or atmosphere
- Ignore live pan/zoom; world units map 1:1 to export pixels (unless max-side clamp scales down)
- Empty board: no download; `alert` with a short message
- Filename: `scratchboard-YYYY-MM-DD.png` (local date)
- Max export side: **8192** pixels (uniform scale-down if needed)
- Client-only; do not change `board.js`, `server.js`, or the WS protocol

## File Structure

| File | Responsibility |
|------|----------------|
| `public/export-png.js` | Pure helpers: bounds, fit size, constants, filename (browser + Node tests) |
| `test/export-png.test.js` | Unit tests for export geometry |
| `public/index.html` | Save dock button |
| `public/app.js` | Gather strokes, offscreen render, download wiring |

**Interfaces:**
- Consumes: nothing
- Produces:
  - `EXPORT_PADDING` → `48`
  - `EXPORT_PAPER` → `'#eef2f6'`
  - `EXPORT_MAX_SIDE` → `8192`
  - `computeContentBounds(strokes, padding = EXPORT_PADDING) → { minX, minY, width, height } | null`
    - `strokes`: array of `{ width: number, points: number[][] }` (at least `width` + `points`)
    - Expand each point by `width / 2`, then add `padding` on all sides
    - Return `null` if no strokes or no points
  - `fitExportSize(width, height, maxSide = EXPORT_MAX_SIDE) → { width, height, scale }`
    - Integers for `width`/`height` ≥ 1; `scale` is the uniform factor applied (1 if no clamp)
  - `exportFilename(date = new Date()) → string` like `scratchboard-2026-09-07.png` (local Y-M-D, zero-padded)

    assert.equal(EXPORT_MAX_SIDE, 8192);

  it('returns null for empty stroke lists', () => {
    assert.equal(computeContentBounds([]), null);
    assert.equal(computeContentBounds([{ width: 5, points: [] }]), null);

  it('pads a single point by half-width plus EXPORT_PADDING', () => {
    const bounds = computeContentBounds([{ width: 10, points: [[100, 200]] }]);
    // half width = 5 → content 95..105, 195..205; +48 pad
    assert.deepEqual(bounds, {
      minX: 95 - 48,
      minY: 195 - 48,
      width: 10 + 96,
      height: 10 + 96,
    });
  });

  it('unions multiple strokes', () => {
    const bounds = computeContentBounds([
      { width: 2, points: [[0, 0], [10, 0]] },
      { width: 4, points: [[50, 80]] },
    ]);
    // stroke1: x -1..11, y -1..1; stroke2: x 48..52, y 78..82
    // union: x -1..52, y -1..82 → +48 pad
    assert.equal(bounds.minX, -1 - 48);
    assert.equal(bounds.minY, -1 - 48);
    assert.equal(bounds.width, 52 - -1 + 96);
    assert.equal(bounds.height, 82 - -1 + 96);
  });

  it('does not scale when under max side', () => {
    assert.deepEqual(fitExportSize(100, 50), { width: 100, height: 50, scale: 1 });
  });

  it('scales down uniformly when over max side', () => {
    const { width, height, scale } = fitExportSize(16000, 8000, 8192);
    assert.equal(scale, 8192 / 16000);
    assert.equal(width, 8192);
    assert.equal(height, 4096);
  });

  it('formats local date filename', () => {
    const d = new Date(2026, 8, 7); // Sep 7, 2026 local
    assert.equal(exportFilename(d), 'scratchboard-2026-09-07.png');
  });

- [ ] **Step 3: Implement `public/export-png.js`**

Create `public/export-png.js`:

```js
export const EXPORT_PADDING = 48;
export const EXPORT_PAPER = '#eef2f6';
export const EXPORT_MAX_SIDE = 8192;

/**
 * @param {{ width: number, points: number[][] }[]} strokes
 * @param {number} [padding]
 * @returns {{ minX: number, minY: number, width: number, height: number } | null}
 */
export function computeContentBounds(strokes, padding = EXPORT_PADDING) {
  let minX = Infinity;
  let minY = Infinity;
  let maxX = -Infinity;
  let maxY = -Infinity;
  let any = false;

  for (const stroke of strokes || []) {
    const half = (stroke.width ?? 0) / 2;
    for (const p of stroke.points || []) {
      if (!Array.isArray(p) || p.length < 2) continue;
      const x = p[0];
      const y = p[1];
      if (!Number.isFinite(x) || !Number.isFinite(y)) continue;
      any = true;
      minX = Math.min(minX, x - half);
      minY = Math.min(minY, y - half);
      maxX = Math.max(maxX, x + half);
      maxY = Math.max(maxY, y + half);
    }
  }

  if (!any) return null;

  return {
    minX: minX - padding,
    minY: minY - padding,
    width: maxX - minX + padding * 2,
    height: maxY - minY + padding * 2,
  };
}

/**
 * @param {number} width
 * @param {number} height
 * @param {number} [maxSide]
 * @returns {{ width: number, height: number, scale: number }}
 */
export function fitExportSize(width, height, maxSide = EXPORT_MAX_SIDE) {
  const w = Math.max(1, width);
  const h = Math.max(1, height);
  const longest = Math.max(w, h);
  if (longest <= maxSide) {
    return {
      width: Math.max(1, Math.round(w)),
      height: Math.max(1, Math.round(h)),
      scale: 1,
    };
  }
  const scale = maxSide / longest;
  return {
    width: Math.max(1, Math.round(w * scale)),
    height: Math.max(1, Math.round(h * scale)),
    scale,
  };
}

/**
 * @param {Date} [date]
 * @returns {string}
 */
export function exportFilename(date = new Date()) {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, '0');
  const d = String(date.getDate()).padStart(2, '0');
  return `scratchboard-${y}-${m}-${d}.png`;
}
```

```js
function gatherExportStrokes() {
  const strokes = strokeHistory.map(cloneStroke);
  for (const entry of remoteStrokes.values()) {
    strokes.push(cloneStroke(entry));
  }
  if (drawing && localStrokePoints.length) {
    strokes.push({
      id: strokeId || 'local-in-progress',
      tool: currentStrokeTool,
      color: currentStrokeColor,
      size: currentStrokeSize,
      width: currentStrokeWidth,
      points: localStrokePoints.map((p) => [p[0], p[1]]),
    });
  }
  return strokes;
}

function drawStrokesOnExportCtx(exportCtx, strokes) {
  exportCtx.lineCap = 'round';
  exportCtx.lineJoin = 'round';
  for (const s of strokes) {
    const tool = s.tool || 'pen';
    const color = s.color || DEFAULT_COLOR;
    const width = s.width ?? DEFAULT_SIZE;
    const points = s.points || [];

    if (tool === 'eraser') {
      exportCtx.globalCompositeOperation = 'destination-out';
      exportCtx.strokeStyle = 'rgba(0,0,0,1)';
      exportCtx.fillStyle = 'rgba(0,0,0,1)';
    } else {
      exportCtx.globalCompositeOperation = 'source-over';
      exportCtx.strokeStyle = color;
      exportCtx.fillStyle = color;
    }
    exportCtx.lineWidth = width;

    if (points.length < 2) {
      if (points.length === 1) {
        exportCtx.beginPath();
        exportCtx.arc(points[0][0], points[0][1], width / 2, 0, Math.PI * 2);
        exportCtx.fill();
      }
    } else {
      exportCtx.beginPath();
      exportCtx.moveTo(points[0][0], points[0][1]);
      for (let i = 1; i < points.length; i++) {
        exportCtx.lineTo(points[i][0], points[i][1]);
      }
      exportCtx.stroke();
    }
    exportCtx.globalCompositeOperation = 'source-over';
  }
}

function saveBoardAsPng() {
  const strokes = gatherExportStrokes();
  const bounds = computeContentBounds(strokes);
  if (!bounds) {
    window.alert('Nothing to save yet — draw something first.');
    return;
  }

  const fitted = fitExportSize(bounds.width, bounds.height);
  const offscreen = document.createElement('canvas');
  offscreen.width = fitted.width;
  offscreen.height = fitted.height;
  const exportCtx = offscreen.getContext('2d');
  if (!exportCtx) {
    window.alert('Could not create export canvas.');
    return;
  }

  exportCtx.fillStyle = EXPORT_PAPER;
  exportCtx.fillRect(0, 0, fitted.width, fitted.height);
  exportCtx.setTransform(
    fitted.scale,
    0,
    0,
    fitted.scale,
    -bounds.minX * fitted.scale,
    -bounds.minY * fitted.scale,
  );
  drawStrokesOnExportCtx(exportCtx, strokes);

  offscreen.toBlob((blob) => {
    if (!blob) {
      window.alert('Could not create PNG.');
      return;
    }
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = exportFilename();
    document.body.appendChild(a);
    a.click();
    a.remove();
    URL.revokeObjectURL(url);
  }, 'image/png');
}
```
